⚡ Sincronização via PowerShell (Home PC)

# Setup inicial (uma vez):
. "D:\Cerebro_Virtual\modulo_github_cerebro_sync.ps1"
Invoke-GitHubCerebroSync -Setup

# Sync automático (todos os KIs):
pwsh "$env:OneDrive\Sync_Cerebro_Global.ps1" -ModoAntigravityBridge -PushAll

# Iniciar bridge server local:
pwsh "D:\Cerebro_Virtual\bridge_server.ps1"
